# 로컬 실행
cd /d d:\psm\coding\the-fallen
npm install
npm run dev



# 로컬 Git 초기화 및 커밋
git init
git add .
git commit -m "Initial scaffold"

# 권장: GitHub CLI로 리포지토리 생성 및 푸시 (gh 인증 필요)
gh repo create bagyeon/the-fallen --public --source=. --remote=origin --push

# 빌드 및 gh-pages 배포
npm run build
npm run deploy


# GitHub Pages 활성화
GitHub 레포 → Settings → Pages
Source: Deploy from a branch
Branch: gh-pages / / (root) 선택 후 Save
